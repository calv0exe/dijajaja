ALGORITMO Dijkstra
    pontos ← ["Rodoviária", "Super Fácil", "Rjr Gesso", "Mercadinho Popular",
              "Contrukasa", "Caixa Econômica", "Arena das Dunas"]

    // As linhas e colunas seguem a ordem de pontos, de 0 a 6.
    // Cada valor é uma distância em metros. Zero indica ausência de ligação direta.
    grafo ← [
        [  0,  630,   0,    0,    0,    0,    0],
        [630,    0, 635,    0,    0, 2600,    0],
        [  0,  635,   0,  500,    0,    0,    0],
        [  0,    0, 500,    0, 2100,    0,    0],
        [  0,    0,   0, 2100,    0,  800, 2100],
        [  0, 2600,   0,    0,  800,    0, 2050],
        [  0,    0,   0,    0, 2100, 2050,    0]
    ]

    origem ← LerPonto(pontos, "Digite o ponto de origem: ")
    SE origem = -1 ENTÃO ENCERRE O ALGORITMO
    destino ← LerPonto(pontos, "Digite o ponto de destino: ")
    SE destino = -1 ENTÃO ENCERRE O ALGORITMO

    PARA i DE 0 ATÉ 6 FAÇA
        distancia[i] ← INFINITO
        visitado[i] ← FALSO
        rota[i] ← ""
    FIM PARA
    distancia[origem] ← 0
    rota[origem] ← pontos[origem]

    ENQUANTO VERDADEIRO FAÇA
        // Escolhe o ponto ainda não visitado com a menor distância desde a origem.
        atual ← -1
        menor ← INFINITO
        PARA i DE 0 ATÉ 6 FAÇA
            SE NÃO visitado[i] E distancia[i] < menor ENTÃO
                atual ← i
                menor ← distancia[i]
            FIM SE
        FIM PARA

        SE atual = -1 OU atual = destino ENTÃO SAIA DO LAÇO
        visitado[atual] ← VERDADEIRO

        // Se passar pelo ponto atual encurtar o caminho, guarda a distância e a rota.
        PARA vizinho DE 0 ATÉ 6 FAÇA
            peso ← grafo[atual][vizinho]
            SE peso > 0 E menor + peso < distancia[vizinho] ENTÃO
                distancia[vizinho] ← menor + peso
                rota[vizinho] ← rota[atual] + " -> " + pontos[vizinho]
            FIM SE
        FIM PARA
    FIM ENQUANTO

    SE rota[destino] = "" ENTÃO
        ESCREVA "Não existe rota entre os dois pontos."
    SENÃO
        ESCREVA "Melhor rota: ", rota[destino]
        ESCREVA "Distância: ", distancia[destino], " m"
    FIM SE
FIM DO ALGORITMO

FUNÇÃO LerPonto(pontos, pergunta)
    ESCREVA "Pontos disponíveis:"
    PARA CADA ponto EM pontos FAÇA
        ESCREVA ponto
    FIM PARA

    ENQUANTO VERDADEIRO FAÇA
        ESCREVA pergunta
        SE não houver outra linha de entrada ENTÃO RETORNE -1
        LEIA a linha inteira em nome
        REMOVA os espaços do início e do fim de nome
        PARA i DE 0 ATÉ 6 FAÇA
            SE nome for igual a pontos[i], ignorando maiúsculas e minúsculas, ENTÃO
                RETORNE i
            FIM SE
        FIM PARA
        ESCREVA "Ponto inválido. Digite um dos nomes apresentados."
    FIM ENQUANTO
FIM DA FUNÇÃO
